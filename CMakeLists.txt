cmake_minimum_required(VERSION 3.16)
project(hash_aggregation LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(MSVC)
  add_compile_options(/W4)
else()
  add_compile_options(-Wall -Wextra -Wpedantic)
endif()
add_executable(demo src/demo.cpp)
add_executable(aggregation_test tests/aggregation_test.cpp)
target_include_directories(aggregation_test PRIVATE src)
enable_testing()
add_test(NAME hash_aggregation COMMAND aggregation_test)
set_tests_properties(hash_aggregation PROPERTIES TIMEOUT 30)
