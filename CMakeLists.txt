cmake_minimum_required(VERSION 3.16)
project(database_topic LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(topic INTERFACE)
target_include_directories(topic INTERFACE ${CMAKE_CURRENT_SOURCE_DIR}/src)
if(MSVC)
  target_compile_options(topic INTERFACE /W4)
else()
  target_compile_options(topic INTERFACE -Wall -Wextra -Wpedantic)
endif()
add_executable(demo src/demo.cpp)
target_link_libraries(demo PRIVATE topic)
add_executable(topic_tests tests/topic_tests.cpp)
target_link_libraries(topic_tests PRIVATE topic)
enable_testing()
add_test(NAME topic_tests COMMAND topic_tests)
