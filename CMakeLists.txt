cmake_minimum_required(VERSION 3.16)
project(external_merge_sort LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(MSVC)
  add_compile_options(/W4)
else()
  add_compile_options(-Wall -Wextra -Wpedantic)
endif()
add_executable(demo src/demo.cpp)
add_executable(external_sort_test tests/external_sort_test.cpp)
target_include_directories(external_sort_test PRIVATE src)
enable_testing()
add_test(NAME external_sort COMMAND external_sort_test)
set_tests_properties(external_sort PROPERTIES TIMEOUT 60)
