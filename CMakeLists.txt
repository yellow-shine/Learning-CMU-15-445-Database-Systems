cmake_minimum_required(VERSION 3.16)
project(tutorial LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_executable(demo src/demo.cpp)
add_executable(tutorial_test tests/test.cpp)
target_include_directories(tutorial_test PRIVATE src)
foreach(target demo tutorial_test)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
  endif()
endforeach()
enable_testing()
add_test(NAME mechanism COMMAND tutorial_test)
set_tests_properties(mechanism PROPERTIES TIMEOUT 10)
