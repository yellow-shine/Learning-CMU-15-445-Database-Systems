cmake_minimum_required(VERSION 3.16)
project(tutorial_join LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(join src/join.cpp)
target_include_directories(join PUBLIC src)
add_executable(demo src/demo.cpp)
target_link_libraries(demo PRIVATE join)
add_executable(join_test tests/join_test.cpp)
target_link_libraries(join_test PRIVATE join)
foreach(target join demo join_test)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4)
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
    endif()
endforeach()
enable_testing()
add_test(NAME join_test COMMAND join_test)
