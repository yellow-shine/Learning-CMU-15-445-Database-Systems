cmake_minimum_required(VERSION 3.16)
project(relational_tutorial LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(tutorial INTERFACE)
target_include_directories(tutorial INTERFACE ${CMAKE_CURRENT_SOURCE_DIR}/src)
if(MSVC)
  target_compile_options(tutorial INTERFACE /W4)
else()
  target_compile_options(tutorial INTERFACE -Wall -Wextra -Wpedantic)
endif()
add_executable(demo src/demo.cpp)
target_link_libraries(demo PRIVATE tutorial)
include(CTest)
if(BUILD_TESTING)
  add_executable(tutorial_tests tests/tests.cpp)
  target_link_libraries(tutorial_tests PRIVATE tutorial)
  add_test(NAME tutorial_tests COMMAND tutorial_tests)
endif()
