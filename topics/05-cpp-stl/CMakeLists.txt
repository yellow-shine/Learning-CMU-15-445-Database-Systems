cmake_minimum_required(VERSION 3.16)
project(database_tutorial LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Threads REQUIRED)
enable_testing()
add_executable(demo src/demo.cpp)
add_executable(topic_tests tests/topic_tests.cpp)
foreach(target demo topic_tests)
  target_include_directories(${target} PRIVATE src)
  target_link_libraries(${target} PRIVATE Threads::Threads)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
  endif()
endforeach()
add_test(NAME topic_tests COMMAND topic_tests)
set_tests_properties(topic_tests PROPERTIES TIMEOUT 20)
