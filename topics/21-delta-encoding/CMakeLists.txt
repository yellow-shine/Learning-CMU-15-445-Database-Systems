cmake_minimum_required(VERSION 3.16)
project(compression_tutorial LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(codec src/codec.cpp)
target_include_directories(codec PUBLIC src)
if(MSVC)
  target_compile_options(codec PUBLIC /W4)
else()
  target_compile_options(codec PUBLIC -Wall -Wextra -Wpedantic -Wconversion)
endif()
add_executable(demo src/demo.cpp)
target_link_libraries(demo PRIVATE codec)
add_executable(codec_test tests/codec_test.cpp)
target_link_libraries(codec_test PRIVATE codec)
enable_testing()
add_test(NAME codec_test COMMAND codec_test)
set_tests_properties(codec_test PROPERTIES TIMEOUT 20)
