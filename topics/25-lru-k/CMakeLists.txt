cmake_minimum_required(VERSION 3.16)
project(25_lru_k LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Threads REQUIRED)
add_library(tutorial INTERFACE)
target_include_directories(tutorial INTERFACE src)
target_link_libraries(tutorial INTERFACE Threads::Threads)
if(MSVC)
  target_compile_options(tutorial INTERFACE /W4)
else()
  target_compile_options(tutorial INTERFACE -Wall -Wextra -Wpedantic)
endif()
add_executable(demo src/demo.cpp)
target_link_libraries(demo PRIVATE tutorial)
add_executable(tutorial_tests tests/tests.cpp)
target_link_libraries(tutorial_tests PRIVATE tutorial)
enable_testing()
add_test(NAME correctness COMMAND tutorial_tests)
set_tests_properties(correctness PROPERTIES TIMEOUT 20)
