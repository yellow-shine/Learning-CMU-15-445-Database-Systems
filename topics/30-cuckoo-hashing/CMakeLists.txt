cmake_minimum_required(VERSION 3.16)
project(hash_tutorial LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_executable(demo src/demo.cpp)
add_executable(hash_tests tests/hash_test.cpp)
target_include_directories(hash_tests PRIVATE src)
foreach(target demo hash_tests)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
  endif()
endforeach()
enable_testing()
add_test(NAME hash_test COMMAND hash_tests)
set_tests_properties(hash_test PROPERTIES TIMEOUT 30)
