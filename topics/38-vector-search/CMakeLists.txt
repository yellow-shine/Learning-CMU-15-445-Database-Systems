cmake_minimum_required(VERSION 3.16)
project(index_tutorial LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Threads REQUIRED)
add_executable(demo src/demo.cpp)
add_executable(index_test tests/index_test.cpp)
foreach(target demo index_test)
  target_include_directories(${target} PRIVATE src)
  target_link_libraries(${target} PRIVATE Threads::Threads)
  if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
  endif()
endforeach()
enable_testing()
add_test(NAME index_test COMMAND index_test)
set_tests_properties(index_test PROPERTIES TIMEOUT 45)
